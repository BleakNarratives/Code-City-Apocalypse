import os

archive_path = "/storage/emulated/0/root_2025_archive.txt"
output_root = "/storage/emulated/0/rebuilt_root_2025"

if not os.path.exists(output_root):
    os.makedirs(output_root)

with open(archive_path, "r", encoding="utf-8", errors="ignore") as f:
    lines = f.readlines()

current_file_path = None
current_content = []

for line in lines:
    if line.startswith("===FILE START: "):
        current_file_path = line.strip()[14:-3]  # Extract path
        current_content = []
    elif line.startswith("===FILE END: "):
        if current_file_path:
            full_path = os.path.join(output_root, current_file_path.lstrip("./"))
            os.makedirs(os.path.dirname(full_path), exist_ok=True)
            with open(full_path, "w", encoding="utf-8", errors="ignore") as out_file:
                out_file.writelines(current_content)
        current_file_path = None
    else:
        if current_file_path:
            current_content.append(line)
